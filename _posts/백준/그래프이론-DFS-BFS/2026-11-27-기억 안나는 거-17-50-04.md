import java.io.*;  
import java.util.ArrayList;  
import java.util.Comparator;  
import java.util.PriorityQueue;  
import java.util.StringTokenizer;  
  
public class Main {  
    static BufferedReader br = new BufferedReader(new InputStreamReader(System.in));  
    static BufferedWriter bw = new BufferedWriter(new OutputStreamWriter(System.out));  
    static StringBuilder sb = new StringBuilder();  
  
    public static void main(String[] args) throws IOException {  
        int N = Integer.parseInt(br.readLine());  
        int M = Integer.parseInt(br.readLine());  
  
        int[] costs = new int[N + 1];  
        for (int i = 0; i <= N; i++) {  
            costs[i] = Integer.MAX_VALUE;  
        }  
  
        StringTokenizer st;  
  
        ArrayList<Node>[] graph = new ArrayList[N + 1];  
        for (int i = 1; i <= N; i++) {  
            graph[i] = new ArrayList<>();  
        }  
  
        PriorityQueue<Node> pq = new PriorityQueue<>(Comparator.comparingInt(o -> o.cost));  
  
        for (int i = 0; i < M; i++) {  
            st = new StringTokenizer(br.readLine());  
            int busStart = Integer.parseInt(st.nextToken());  
            int busEnd = Integer.parseInt(st.nextToken());  
            int busCost = Integer.parseInt(st.nextToken());  
  
            graph[busStart].add(new Node(busEnd, busCost));  
        }  
  
        st = new StringTokenizer(br.readLine());  
        int startPoint = Integer.parseInt(st.nextToken());  
        int endPoint = Integer.parseInt(st.nextToken());  
  
        costs[startPoint] = 0;  
        pq.add(new Node(startPoint, 0));  
  
        while (!pq.isEmpty()) {  
            Node node = pq.poll();  
            int tempCost = node.cost;  
            int tempDest = node.dest;  
  
            if (costs[node.dest] < tempCost) {  
                continue;  
            }  
  
            for (Node tempNode : graph[tempDest]) {  
                int dest = tempNode.dest;  
                int cost = tempNode.cost;  
  
                int newCost = tempCost + cost;  
  
                if (newCost < costs[dest]) {  
                    costs[dest] = newCost;  
                    pq.add(new Node(dest, newCost));  
                }  
            }  
        }  
  
        sb.append(costs[endPoint]);  
        bw.write(sb.toString());  
        bw.flush();  
        bw.close();  
        br.close();  
    }  
  
    static class Node {  
        int dest;  
        int cost;  
  
        Node(int dest, int cost) {  
            this.dest = dest;  
            this.cost = cost;  
        }  
    }  
}
